function(ztk_add_test name)
    add_executable(${name} ${name}.c)
    target_link_libraries(${name} PRIVATE ztk)
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES LABELS "unit" TIMEOUT 120)
endfunction()
